# Локальный учебный стенд QA

> [!NOTE]
> QA Lab — небольшое приложение задач для практики ручного тестирования, Postman и Playwright. Сервер запускается локально на Node.js без сторонних пакетов. Контракт намеренно простой; данные живут в памяти, а два фиксированных токена заменяют настоящую аутентификацию.

## Вопрос

Как получить воспроизводимый продукт для учебного тестирования, если нет рабочего проекта и доступа к стендам?

## Контракт

| Запрос | Условие | Результат |
| --- | --- | --- |
| GET `/health` | без авторизации | 200, `{"status":"ok"}` |
| GET `/` | без авторизации | HTML-страница пользователя A |
| GET `/tasks` | Bearer `qa-a` или `qa-b` | 200, массив только своих задач |
| POST `/tasks` | JSON с title | 201, `{id, title, done:false}` |
| GET `/tasks/:id` | свой объект | 200, объект задачи |
| PATCH `/tasks/:id` | свой объект, JSON `{done:boolean}` | 200, обновлённая задача |
| DELETE `/tasks/:id` | свой объект | 204, пустое тело |

Без известного токена `/tasks` возвращает 401. Чужой или отсутствующий ID — 404. Название — строка: после `trim()` длина 1–80 **UTF-16 code units**, как у JavaScript `string.length`. Неверное название — 400 с `error: "title must contain 1..80 characters"`. Повторы POST создают новые задачи: идемпотентность здесь не реализована. PATCH требует только поле done; лишние поля отклоняются.

## Запуск

Создай отдельную папку `qa-lab`, сохрани следующий блок как `qa-lab.mjs`. Нужен Node.js 22+; внешние npm-пакеты не требуются.

```js
// qa-lab.mjs
import http from 'node:http';
import { randomUUID } from 'node:crypto';

const tasks = new Map();
const owners = new Map([['qa-a', 'A'], ['qa-b', 'B']]);
const maxTitleLength = process.env.QA_BUG === '1' ? 81 : 80;

function json(res, status, body) {
  res.writeHead(status, {
    'Content-Type': 'application/json; charset=utf-8',
    'Cache-Control': 'no-store',
  });
  res.end(JSON.stringify(body));
}

function view(task) {
  return { id: task.id, title: task.title, done: task.done };
}

async function readJson(req) {
  if (!req.headers['content-type']?.includes('application/json')) {
    throw new Error('Content-Type must be application/json');
  }
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > 16384) throw new Error('body too large');
    chunks.push(chunk);
  }
  const text = Buffer.concat(chunks).toString('utf8');
  const body = JSON.parse(text);
  if (!body || typeof body !== 'object' || Array.isArray(body)) {
    throw new Error('body must be an object');
  }
  return body;
}

const html = String.raw`<!doctype html>
<html lang="ru">
<meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>QA Lab</title>
<main>
  <h1>Задачи пользователя A</h1>
  <form id="form" novalidate>
    <label for="title">Название</label>
    <input id="title" name="title">
    <button type="submit">Добавить</button>
  </form>
  <p role="alert" id="error"></p>
  <p role="status" id="status">Загрузка</p>
  <ul id="list" aria-label="Задачи"></ul>
</main>
<script>
const form = document.querySelector('#form');
const input = document.querySelector('#title');
const list = document.querySelector('#list');
const error = document.querySelector('#error');
const status = document.querySelector('#status');

async function api(path, options = {}) {
  const response = await fetch(path, {
    ...options,
    headers: {
      Authorization: 'Bearer qa-a',
      'Content-Type': 'application/json',
    },
  });
  if (response.status === 204) return null;
  const body = await response.json();
  if (!response.ok) throw new Error(body.error || 'Request failed');
  return body;
}

async function load() {
  const tasks = await api('/tasks');
  list.replaceChildren();
  for (const task of tasks) {
    const item = document.createElement('li');
    item.dataset.testid = 'task-' + task.id;
    const label = document.createElement('label');
    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.checked = task.done;
    label.append(checkbox, document.createTextNode(task.title));
    checkbox.addEventListener('change', async () => {
      try {
        await api('/tasks/' + task.id, {
          method: 'PATCH', body: JSON.stringify({ done: checkbox.checked }),
        });
        await load();
      } catch (e) { error.textContent = e.message; }
    });
    const remove = document.createElement('button');
    remove.type = 'button';
    remove.textContent = 'Удалить';
    remove.addEventListener('click', async () => {
      try {
        await api('/tasks/' + task.id, { method: 'DELETE' });
        await load();
      } catch (e) { error.textContent = e.message; }
    });
    item.append(label, ' ', remove);
    list.append(item);
  }
  status.textContent = 'Готово';
}

form.addEventListener('submit', async event => {
  event.preventDefault();
  error.textContent = '';
  const button = form.querySelector('button');
  button.disabled = true;
  try {
    await api('/tasks', {
      method: 'POST', body: JSON.stringify({ title: input.value }),
    });
    input.value = '';
    await load();
  } catch (e) { error.textContent = e.message; }
  finally { button.disabled = false; }
});
load().catch(e => { error.textContent = e.message; });
</script>
</html>`;

const server = http.createServer(async (req, res) => {
  try {
    const path = new URL(req.url, 'http://127.0.0.1').pathname;
    if (req.method === 'GET' && path === '/health') {
      return json(res, 200, { status: 'ok' });
    }
    if (req.method === 'GET' && path === '/') {
      res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
      return res.end(html);
    }
    if (path !== '/tasks' && !path.startsWith('/tasks/')) {
      return json(res, 404, { error: 'not found' });
    }
    const auth = req.headers.authorization ?? '';
    const owner = owners.get(auth.startsWith('Bearer ') ? auth.slice(7) : '');
    if (!owner) return json(res, 401, { error: 'unauthorized' });

    if (path === '/tasks' && req.method === 'GET') {
      return json(res, 200, [...tasks.values()]
        .filter(task => task.owner === owner).map(view));
    }
    if (path === '/tasks' && req.method === 'POST') {
      const body = await readJson(req);
      const title = typeof body.title === 'string' ? body.title.trim() : '';
      if (!title || title.length > maxTitleLength) {
        return json(res, 400, { error: 'title must contain 1..80 characters' });
      }
      const task = { id: randomUUID(), owner, title, done: false };
      tasks.set(task.id, task);
      return json(res, 201, view(task));
    }
    if (path === '/tasks') return json(res, 405, { error: 'method not allowed' });

    const id = path.slice('/tasks/'.length);
    const task = tasks.get(id);
    if (!task || task.owner !== owner) {
      return json(res, 404, { error: 'not found' });
    }
    if (req.method === 'GET') return json(res, 200, view(task));
    if (req.method === 'PATCH') {
      const body = await readJson(req);
      if (typeof body.done !== 'boolean' || Object.keys(body).some(k => k !== 'done')) {
        return json(res, 400, { error: 'only boolean done is allowed' });
      }
      task.done = body.done;
      return json(res, 200, view(task));
    }
    if (req.method === 'DELETE') {
      tasks.delete(id);
      res.writeHead(204);
      return res.end();
    }
    return json(res, 405, { error: 'method not allowed' });
  } catch (error) {
    return json(res, 400, { error: error.message });
  }
});

server.listen(Number(process.env.PORT ?? 3007), '127.0.0.1', () => {
  console.log('QA Lab: http://127.0.0.1:' + server.address().port);
});
```

```bash
node qa-lab.mjs
```

Открой `http://127.0.0.1:3007`. Остановить сервер можно Ctrl+C. После перезапуска задачи исчезнут. Если порт занят, сначала найди запущенный экземпляр или выбери другой PORT и измени baseUrl клиентов. В production такой сервер не используют: здесь нет настоящего входа, БД и полного контроля ошибок.

## Учебный дефект

Основной режим соответствует контракту. Для упражнения с баг-репортом останови сервер и в PowerShell запусти режим с намеренной ошибкой границы:

```powershell
$env:QA_BUG = '1'
node qa-lab.mjs
```

Теперь 81 символ ошибочно принимается. После упражнения останови сервер, выполни `Remove-Item Env:\QA_BUG` и запусти обычный режим. Это заранее внедрённый учебный дефект, а не случайно найденная проблема реального продукта.

## Мини-шпаргалка

Один стенд → явный контракт → UI и API → два владельца → управляемый дефект → повторяемая практика.
