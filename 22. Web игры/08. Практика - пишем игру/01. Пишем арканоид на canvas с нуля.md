# Пишем арканоид на canvas с нуля

> [!NOTE]
> Арканоид (Breakout) — идеальная первая игра: в ней есть всё базовое — игровой цикл с `dt`, ввод с клавиатуры и мыши, движение и отскоки, столкновения круга с прямоугольником, удаление объектов, счёт, жизни и состояния игры (старт, игра, пауза, победа, проигрыш). Код ниже — полный рабочий пример в одном файле: ~200 строк без библиотек. Архитектура та же, что в больших играх: **состояние** (мяч, ракетка, кирпичи) отдельно, `update(dt)` меняет его, `render()` рисует, а цикл на `requestAnimationFrame` связывает всё вместе.

## Вопрос

Как собрать простую законченную игру на canvas: от игрового цикла до победы и проигрыша?

## Шаг 1. Константы и состояние

```ts
const canvas = document.querySelector<HTMLCanvasElement>('#game')!;
const ctx = canvas.getContext('2d')!;
const W = canvas.width;
const H = canvas.height;

type State = 'ready' | 'playing' | 'paused' | 'won' | 'lost';

const paddle = { x: W / 2 - 50, y: H - 30, w: 100, h: 12, speed: 520 };
const ball = { x: 0, y: 0, r: 7, vx: 0, vy: 0, speed: 360 };

interface Brick { x: number; y: number; w: number; h: number; hp: number; color: string }
let bricks: Brick[] = [];

let state: State = 'ready';
let score = 0;
let lives = 3;
```

## Шаг 2. Уровень

```ts
const ROW_COLORS = ['#ef4444', '#f97316', '#eab308', '#22c55e', '#3b82f6'];

function buildLevel() {
  const cols = 10, rows = 5, gap = 6, top = 50;
  const bw = (W - gap * (cols + 1)) / cols;
  const bh = 18;
  bricks = [];
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      bricks.push({
        x: gap + c * (bw + gap),
        y: top + r * (bh + gap),
        w: bw,
        h: bh,
        hp: r === 0 ? 2 : 1,           // верхний ряд — прочнее
        color: ROW_COLORS[r],
      });
    }
  }
}

function resetBall() {
  ball.x = paddle.x + paddle.w / 2;
  ball.y = paddle.y - ball.r - 1;
  ball.vx = 0;
  ball.vy = 0;
}

function launchBall() {
  const angle = -Math.PI / 2 + (Math.random() - 0.5) * 0.6; // вверх с небольшим разбросом
  ball.vx = Math.cos(angle) * ball.speed;
  ball.vy = Math.sin(angle) * ball.speed;
}
```

## Шаг 3. Ввод

```ts
const keys = new Set<string>();
let pointerX: number | null = null;

addEventListener('keydown', (e) => {
  if (['ArrowLeft', 'ArrowRight', 'Space'].includes(e.code)) e.preventDefault();
  keys.add(e.code);

  if (e.code === 'Space') onAction();
  if (e.code === 'KeyP' || e.code === 'Escape') {
    if (state === 'playing') state = 'paused';
    else if (state === 'paused') state = 'playing';
  }
});
addEventListener('keyup', (e) => keys.delete(e.code));
addEventListener('blur', () => { keys.clear(); if (state === 'playing') state = 'paused'; });

canvas.addEventListener('pointermove', (e) => {
  const r = canvas.getBoundingClientRect();
  pointerX = ((e.clientX - r.left) / r.width) * W;   // экран → координаты игры
});
canvas.addEventListener('pointerdown', onAction);

function onAction() {
  if (state === 'ready') { state = 'playing'; launchBall(); }
  else if (state === 'won' || state === 'lost') newGame();
}
```

## Шаг 4. Обновление

```ts
const clamp = (v: number, min: number, max: number) => Math.max(min, Math.min(max, v));

function circleRect(c: typeof ball, r: { x: number; y: number; w: number; h: number }) {
  const nx = clamp(c.x, r.x, r.x + r.w);
  const ny = clamp(c.y, r.y, r.y + r.h);
  return (c.x - nx) ** 2 + (c.y - ny) ** 2 < c.r ** 2;
}

function update(dt: number) {
  // ракетка: клавиатура или мышь
  const dir = Number(keys.has('ArrowRight')) - Number(keys.has('ArrowLeft'));
  if (dir !== 0) {
    paddle.x += dir * paddle.speed * dt;
    pointerX = null;                      // клавиатура «перехватывает» управление
  } else if (pointerX !== null) {
    paddle.x = pointerX - paddle.w / 2;
  }
  paddle.x = clamp(paddle.x, 0, W - paddle.w);

  if (state === 'ready') { resetBall(); return; }   // мяч «лежит» на ракетке
  if (state !== 'playing') return;

  // подшаги: мяч не проскочит сквозь кирпич при лаге
  const steps = Math.ceil((Math.hypot(ball.vx, ball.vy) * dt) / ball.r);
  for (let i = 0; i < steps; i++) step(dt / steps);
}

function step(dt: number) {
  ball.x += ball.vx * dt;
  ball.y += ball.vy * dt;

  // стены
  if (ball.x - ball.r < 0) { ball.x = ball.r; ball.vx = Math.abs(ball.vx); }
  if (ball.x + ball.r > W) { ball.x = W - ball.r; ball.vx = -Math.abs(ball.vx); }
  if (ball.y - ball.r < 0) { ball.y = ball.r; ball.vy = Math.abs(ball.vy); }

  // упал вниз
  if (ball.y - ball.r > H) {
    lives--;
    state = lives > 0 ? 'ready' : 'lost';
    return;
  }

  // ракетка: угол отскока зависит от точки удара
  if (ball.vy > 0 && circleRect(ball, paddle)) {
    const hit = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2); // -1..1
    const angle = clamp(hit, -1, 1) * (Math.PI / 3);                    // до ±60°
    const speed = Math.hypot(ball.vx, ball.vy);
    ball.vx = Math.sin(angle) * speed;
    ball.vy = -Math.cos(angle) * speed;
    ball.y = paddle.y - ball.r;
  }

  // кирпичи
  for (const b of bricks) {
    if (b.hp <= 0 || !circleRect(ball, b)) continue;

    // с какой стороны удар: сравниваем перекрытия
    const overlapX = Math.min(ball.x + ball.r - b.x, b.x + b.w - (ball.x - ball.r));
    const overlapY = Math.min(ball.y + ball.r - b.y, b.y + b.h - (ball.y - ball.r));
    if (overlapX < overlapY) ball.vx = -ball.vx;
    else ball.vy = -ball.vy;

    b.hp--;
    score += 10;
    if (b.hp <= 0) {
      score += 40;
      ball.vx *= 1.02;                    // игра постепенно ускоряется
      ball.vy *= 1.02;
    }
    break;                                // не больше одного кирпича за подшаг
  }

  bricks = bricks.filter((b) => b.hp > 0);
  if (bricks.length === 0) state = 'won';
}
```

## Шаг 5. Отрисовка

```ts
function render() {
  ctx.fillStyle = '#0f172a';
  ctx.fillRect(0, 0, W, H);

  for (const b of bricks) {
    ctx.fillStyle = b.color;
    ctx.globalAlpha = b.hp > 1 ? 1 : 0.85;
    ctx.beginPath();
    ctx.roundRect(b.x, b.y, b.w, b.h, 4);
    ctx.fill();
  }
  ctx.globalAlpha = 1;

  ctx.fillStyle = '#e2e8f0';
  ctx.beginPath();
  ctx.roundRect(paddle.x, paddle.y, paddle.w, paddle.h, 6);
  ctx.fill();

  ctx.fillStyle = '#fbbf24';
  ctx.beginPath();
  ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
  ctx.fill();

  // HUD
  ctx.fillStyle = '#94a3b8';
  ctx.font = '16px system-ui, sans-serif';
  ctx.textBaseline = 'top';
  ctx.textAlign = 'left';
  ctx.fillText(`Счёт: ${score}`, 12, 12);
  ctx.textAlign = 'right';
  ctx.fillText('♥'.repeat(lives), W - 12, 12);

  const messages: Partial<Record<State, string>> = {
    ready: 'Пробел или клик — запуск',
    paused: 'Пауза (P)',
    won: 'Победа! Клик — заново',
    lost: 'Игра окончена. Клик — заново',
  };
  const msg = messages[state];
  if (msg) {
    ctx.fillStyle = 'rgb(2 6 23 / 0.6)';
    ctx.fillRect(0, H / 2 - 30, W, 60);
    ctx.fillStyle = 'white';
    ctx.font = 'bold 22px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(msg, W / 2, H / 2);
  }
}
```

## Шаг 6. Цикл и запуск

```ts
function newGame() {
  score = 0;
  lives = 3;
  buildLevel();
  state = 'ready';
}

let last = performance.now();

function loop(now: number) {
  const dt = Math.min((now - last) / 1000, 0.05);
  last = now;
  update(dt);
  render();
  requestAnimationFrame(loop);
}

newGame();
requestAnimationFrame(loop);
```

```ts
const best = Number(localStorage.getItem('breakout:best') ?? 0);
if (score > best) localStorage.setItem('breakout:best', String(score));
```

## Частые ошибки

- Отражать скорость мяча от ракетки, когда он летит вверх, — мяч «залипает» внутри. Проверка `ball.vy > 0`.
- Разрушать несколько кирпичей за один шаг и инвертировать скорость дважды — мяч продолжает лететь сквозь стену кирпичей.
- Не ограничивать `dt` — после переключения вкладки мяч оказывается за пределами поля.
- Считать координаты мыши без учёта CSS-масштаба холста.

## Что важно помнить

- Игра = состояние + `update(dt)` + `render()` + цикл.
- Состояния игры (`ready`, `playing`, `paused`, `won`, `lost`) упрощают логику и экраны.
- Столкновение круга с прямоугольником через ближайшую точку; сторона удара — по меньшему перекрытию.
- Угол отскока от ракетки зависит от точки удара — так игрок управляет мячом.

## Мини-шпаргалка

```ts
update(dt) → ввод → движение → стены → ракетка → кирпичи → победа/проигрыш
render()   → фон → кирпичи → ракетка → мяч → HUD → сообщение состояния
```
